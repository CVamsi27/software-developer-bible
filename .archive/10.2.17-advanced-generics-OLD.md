# Advanced TypeScript Generics — Conditional Types, Mapped Types, Template Literals

## 1. Conditional Types

### Basic Conditional Type

```typescript
// If T is a string, return number; otherwise return boolean
type IsString<T> = T extends string ? number : boolean;

type A = IsString<string>;  // number
type B = IsString<number>;  // boolean
```

### Extracting Types

```typescript
// Extract the type of a Promise's resolved value
type Unwrap<T> = T extends Promise<infer U> ? U : T;

type A = Unwrap<Promise<string>>;  // string
type B = Unwrap<Promise<number>>;  // number
type C = Unwrap<string>;           // string (not a Promise)

// Extract array element type
type ElementOf<T> = T extends (infer U)[] ? U : never;

type A = ElementOf<string[]>;  // string
type B = ElementOf<number[]>;  // number
```

### Distributive Conditional Types

```typescript
// Distributes over union types
type ToArray<T> = T extends any ? T[] : never;

type A = ToArray<string | number>;  // string[] | number[]
// Equivalent to: ToArray<string> | ToArray<number>

// Prevent distribution with square brackets
type ToArrayNonDist<T> = [T] extends [any] ? T[] : never;

type B = ToArrayNonDist<string | number>;  // (string | number)[]
```

## 2. Mapped Types

### Basic Mapped Type

```typescript
// Make all properties optional
type MyPartial<T> = {
  [K in keyof T]?: T[K];
};

// Make all properties required
type MyRequired<T> = {
  [K in keyof T]-?: T[K];
};

// Make all properties readonly
type MyReadonly<T> = {
  readonly [K in keyof T]: T[K];
};
```

### Key Remapping

```typescript
// Prefix all keys with 'get'
type Getters<T> = {
  [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
};

interface Patient {
  id: string;
  name: string;
  email: string;
}

type PatientGetters = Getters<Patient>;
// {
//   getId: () => string;
//   getName: () => string;
//   getEmail: () => string;
// }

// Filter keys by value type
type StringKeys<T> = {
  [K in keyof T as T[K] extends string ? K : never]: T[K];
};

type PatientStrings = StringKeys<Patient>;
// { id: string; name: string; email: string; }
```

## 3. Template Literal Types

### Basic Template Literals

```typescript
type EventName = 'click' | 'focus' | 'blur';
type HandlerName = `on${Capitalize<EventName>}`;
// 'onClick' | 'onFocus' | 'onBlur'

type HTTPMethod = 'GET' | 'POST' | 'PUT' | 'DELETE';
type Route = `/api/${string}`;
type Endpoint = `${HTTPMethod} ${Route}`;
// 'GET /api/...' | 'POST /api/...' | etc.
```

### Practical Example: Type-Safe Event Emitters

```typescript
type EventMap = {
  'appointment:created': { id: string; patientId: string };
  'appointment:cancelled': { id: string; reason: string };
  'notification:sent': { userId: string; channel: string };
};

// Extract event name type
type EventName = keyof EventMap;

// Extract payload type for a specific event
type EventPayload<K extends EventName> = EventMap[K];

class TypedEventEmitter {
  private handlers: Map<string, Function[]> = new Map();
  
  on<K extends EventName>(
    event: K,
    handler: (payload: EventPayload<K>) => void
  ): void {
    const handlers = this.handlers.get(event) || [];
    handlers.push(handler);
    this.handlers.set(event, handlers);
  }
  
  emit<K extends EventName>(event: K, payload: EventPayload<K>): void {
    const handlers = this.handlers.get(event) || [];
    handlers.forEach(h => h(payload));
  }
}

// Usage — fully type-safe
const emitter = new TypedEventEmitter();

emitter.on('appointment:created', (payload) => {
  console.log(payload.id);      // ✅ string
  console.log(payload.patientId); // ✅ string
});

emitter.emit('appointment:created', {
  id: 'abc',
  patientId: 'def',
  // @ts-expect-error — extra property not allowed
  extra: 'not allowed',
});
```

## 4. Type Inference

### infer in Function Signatures

```typescript
// Extract function return type
type ReturnOf<T> = T extends (...args: any[]) => infer R ? R : never;

type A = ReturnOf<() => string>;  // string
type B = ReturnOf<(x: number) => boolean>;  // boolean

// Extract function parameter types
type ParamsOf<T> = T extends (...args: infer P) => any ? P : never;

type A = ParamsOf<(x: number, y: string) => void>;  // [number, string]
```

### infer in Conditional Chains

```typescript
// Deep unwrapping
type DeepUnwrap<T> =
  T extends Promise<infer U> ? DeepUnwrap<U> :
  T extends Array<infer U> ? DeepUnwrap<U>[] :
  T;

type A = DeepUnwrap<Promise<string[]>>;  // string[]
type B = DeepUnwrap<Promise<Promise<number>>>;  // number
```

## 5. Practical Examples

### Type-Safe API Client

```typescript
interface APIRoutes {
  'GET /patients/:id': {
    response: Patient;
    params: { id: string };
  };
  'POST /patients': {
    response: Patient;
    body: CreatePatientDto;
  };
  'GET /appointments': {
    response: Appointment[];
    query: { clinicId: string; date?: string };
  };
}

type ExtractMethod<T> = T extends `${infer M} ${string}` ? M : never;
type ExtractPath<T> = T extends `${string} ${infer P}` ? P : never;

type APIResponse<T extends keyof APIRoutes> = APIRoutes[T]['response'];

async function apiGet<T extends keyof APIRoutes>(
  route: T
): Promise<APIResponse<T>> {
  const response = await fetch(route);
  return response.json();
}

// Usage — fully type-safe
const patient = await apiGet('GET /patients/:id');
// Type: Patient

const appointments = await apiGet('GET /appointments');
// Type: Appointment[]
```

### Type-Safe Builder Pattern

```typescript
interfaceQueryBuilder<T extends Record<string, any>> {
  private filters: Partial<T> = {};
  
  where<K extends keyof T>(key: K, value: T[K]): this {
    this.filters[key] = value;
    return this;
  }
  
  build(): Partial<T> {
    return this.filters;
  }
}

// Usage
interface PatientFilter {
  clinicId: string;
  name?: string;
  email?: string;
}

const query = new QueryBuilder<PatientFilter>()
  .where('clinicId', '123')  // ✅ string required
  .where('name', 'John')     // ✅ string optional
  // @ts-expect-error — number not assignable to string
  .where('name', 123)
  .build();
```

## Interview Tips

- **Q: What are conditional types?**
  - Types that choose between two types based on a condition
  - Like ternary operators for types
  - Can distribute over union types

- **Q: What are mapped types?**
  - Types that transform existing types by iterating over keys
  - Used for Partial, Required, Readonly, Pick, Omit

- **Q: What are template literal types?**
  - Types that construct strings from other string types
  - Used for type-safe event names, routes, HTTP methods

- **Q: What is `infer`?**
  - Keyword to extract types within conditional types
  - Used to infer return types, parameter types, Promise values

# Rate Limiting — Algorithms, Implementation, Distributed

## Rate Limiting Algorithms

### Fixed Window
```typescript
class FixedWindowRateLimiter {
  private windows: Map<string, { count: number; resetAt: number }> = new Map();

  async isAllowed(key: string, limit: number, windowMs: number): Promise<boolean> {
    const now = Date.now();
    const window = this.windows.get(key);

    if (!window || now > window.resetAt) {
      // New window
      this.windows.set(key, { count: 1, resetAt: now + windowMs });
      return true;
    }

    if (window.count >= limit) {
      return false; // Rate limited
    }

    window.count++;
    return true;
  }
}
```
**Pros:** Simple, memory-efficient  
**Cons:** Boundary spike problem (2x burst at window edge)

### Sliding Window Log
```typescript
class SlidingWindowLog {
  private logs: Map<string, number[]> = new Map();

  async isAllowed(key: string, limit: number, windowMs: number): Promise<boolean> {
    const now = Date.now();
    const windowStart = now - windowMs;

    if (!this.logs.has(key)) {
      this.logs.set(key, []);
    }

    const log = this.logs.get(key)!;
    
    // Remove old entries
    const validEntries = log.filter(timestamp => timestamp > windowStart);
    this.logs.set(key, validEntries);

    if (validEntries.length >= limit) {
      return false; // Rate limited
    }

    validEntries.push(now);
    return true;
  }
}
```
**Pros:** Accurate, no boundary issues  
**Cons:** Memory-intensive (stores all timestamps)

### Sliding Window Counter
```typescript
class SlidingWindowCounter {
  async isAllowed(
    key: string, 
    limit: number, 
    windowMs: number
  ): Promise<{ allowed: boolean; remaining: number; resetAt: number }> {
    const now = Date.now();
    const currentWindow = Math.floor(now / windowMs);
    const previousWindow = currentWindow - 1;

    // Get counts from Redis
    const currentCount = await this.redis.get(`${key}:${currentWindow}`) || 0;
    const previousCount = await this.redis.get(`${key}:${previousWindow}`) || 0;

    // Weighted average
    const elapsed = now % windowMs;
    const weight = 1 - (elapsed / windowMs);
    const estimatedCount = (previousCount * weight) + parseInt(currentCount);

    if (estimatedCount >= limit) {
      return {
        allowed: false,
        remaining: 0,
        resetAt: (currentWindow + 1) * windowMs
      };
    }

    // Increment current window
    await this.redis.multi()
      .incr(`${key}:${currentWindow}`)
      .expire(`${key}:${currentWindow}`, Math.ceil(windowMs / 1000) * 2)
      .exec();

    return {
      allowed: true,
      remaining: Math.max(0, limit - estimatedCount - 1),
      resetAt: (currentWindow + 1) * windowMs
    };
  }
}
```
**Pros:** Good balance of accuracy and memory  
**Cons:** Approximate (weighted average)

### Token Bucket
```typescript
class TokenBucket {
  private tokens: number;
  private lastRefill: number;

  constructor(
    private capacity: number,
    private refillRate: number, // tokens per second
    private lastRefillTime: number = Date.now()
  ) {
    this.tokens = capacity;
    this.lastRefill = lastRefillTime;
  }

  async isAllowed(tokens: number = 1): Promise<boolean> {
    this.refill();

    if (this.tokens >= tokens) {
      this.tokens -= tokens;
      return true;
    }

    return false;
  }

  private refill(): void {
    const now = Date.now();
    const elapsed = (now - this.lastRefill) / 1000;
    const tokensToAdd = elapsed * this.refillRate;

    this.tokens = Math.min(this.capacity, this.tokens + tokensToAdd);
    this.lastRefill = now;
  }
}
```
**Pros:** Allows bursts up to capacity, smooth rate limiting  
**Cons:** More complex to implement

### Leaky Bucket
```typescript
class LeakyBucket {
  private queue: number[] = [];
  private lastLeak: number = Date.now();

  constructor(
    private capacity: number,
    private leakRate: number // requests per second
  ) {}

  async isAllowed(): Promise<boolean> {
    this.leak();

    if (this.queue.length < this.capacity) {
      this.queue.push(Date.now());
      return true;
    }

    return false; // Queue full
  }

  private leak(): void {
    const now = Date.now();
    const elapsed = (now - this.lastLeak) / 1000;
    const leaked = Math.floor(elapsed * this.leakRate);

    this.queue.splice(0, leaked);
    this.lastLeak = now;
  }
}
```
**Pros:** Smooths traffic, predictable output rate  
**Cons:** Adds latency (requests wait in queue)

## Algorithm Comparison

| Algorithm | Accuracy | Memory | Burst Handling | Use Case |
|-----------|----------|--------|----------------|----------|
| Fixed Window | Low | Low | Bad (boundary spike) | Simple APIs |
| Sliding Window Log | High | High | Good | Financial APIs |
| Sliding Window Counter | Medium | Medium | Good | Most APIs |
| Token Bucket | High | Low | Excellent (allows bursts) | APIs with burst traffic |
| Leaky Bucket | High | Medium | Good (smooths) | Queue-based systems |

## Distributed Rate Limiting (Redis)

```typescript
// Redis-based rate limiting with sliding window
async function checkRateLimit(
  redis: Redis,
  key: string,
  limit: number,
  windowMs: number
): Promise<{ allowed: boolean; remaining: number; retryAfter?: number }> {
  const now = Date.now();
  const windowStart = now - windowMs;

  // Lua script for atomic operation
  const script = `
    local key = KEYS[1]
    local window_start = ARGV[1]
    local limit = tonumber(ARGV[2])
    local now = ARGV[3]
    
    -- Remove old entries
    redis.call('ZREMRANGEBYSCORE', key, 0, window_start)
    
    -- Count current window
    local current = redis.call('ZCARD', key)
    
    if current < limit then
      -- Add new request
      redis.call('ZADD', key, now, now .. math.random())
      redis.call('PEXPIRE', key, ARGV[4])
      return {1, limit - current - 1}
    else
      -- Get oldest entry for retry-after
      local oldest = redis.call('ZRANGE', key, 0, 0, 'WITHSCORES')
      local retry_after = math.ceil((tonumber(oldest[2]) + ARGV[4] - now) / 1000)
      return {0, 0, retry_after}
    end
  `;

  const result = await redis.eval(
    script,
    1,
    key,
    windowStart.toString(),
    limit.toString(),
    now.toString(),
    windowMs.toString()
  );

  return {
    allowed: result[0] === 1,
    remaining: result[1],
    retryAfter: result[2] ? parseInt(result[2]) : undefined
  };
}
```

### Response Headers
```typescript
// Standard rate limit headers
function setRateLimitHeaders(
  res: Response,
  limit: number,
  remaining: number,
  resetAt: number
): void {
  res.setHeader('X-RateLimit-Limit', limit);
  res.setHeader('X-RateLimit-Remaining', remaining);
  res.setHeader('X-RateLimit-Reset', Math.ceil(resetAt / 1000));
}

// When rate limited
function setRateLimitExceeded(
  res: Response,
  retryAfter: number
): void {
  res.setHeader('Retry-After', retryAfter);
  res.status(429).json({
    error: 'Too Many Requests',
    message: `Rate limit exceeded. Retry after ${retryAfter} seconds.`,
    retryAfter
  });
}
```

## Per-Tenant Rate Limiting (Docita)

```typescript
// Multi-tenant rate limiting
const rateLimits = {
  // Per clinic
  clinic: {
    'appointments:create': { limit: 100, window: '1h' },
    'patients:read': { limit: 1000, window: '1h' },
    'reports:generate': { limit: 10, window: '1d' }
  },
  
  // Per user
  user: {
    'appointments:create': { limit: 20, window: '1h' },
    'patients:read': { limit: 200, window: '1h' }
  },
  
  // Global
  global: {
    'auth:login': { limit: 10, window: '15m' },
    'password:reset': { limit: 3, window: '1h' }
  }
};

// Rate limit middleware
async function rateLimitMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  const userId = req.user?.id;
  const clinicId = req.user?.clinicId;
  const route = `${req.method}:${req.route.path}`;

  // Check user limit first
  const userKey = `ratelimit:user:${userId}:${route}`;
  const userLimit = rateLimits.user[route];
  
  if (userLimit) {
    const result = await checkRateLimit(redis, userKey, userLimit.limit, parseDuration(userLimit.window));
    if (!result.allowed) {
      setRateLimitExceeded(res, result.retryAfter!);
      return;
    }
    setRateLimitHeaders(res, userLimit.limit, result.remaining, result.resetAt);
  }

  // Then check clinic limit
  const clinicKey = `ratelimit:clinic:${clinicId}:${route}`;
  const clinicLimit = rateLimits.clinic[route];
  
  if (clinicLimit) {
    const result = await checkRateLimit(redis, clinicKey, clinicLimit.limit, parseDuration(clinicLimit.window));
    if (!result.allowed) {
      setRateLimitExceeded(res, result.retryAfter!);
      return;
    }
  }

  next();
}
```

## NestJS Exception Filter

```typescript
// Global exception filter
@Catch()
export class GlobalExceptionFilter implements ExceptionFilter {
  catch(exception: unknown, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const request = ctx.getRequest<Request>();

    // Generate request ID for tracing
    const requestId = request.headers['x-request-id'] || uuid();

    // Handle known exceptions
    if (exception instanceof HttpException) {
      const status = exception.getStatus();
      const exceptionResponse = exception.getResponse();

      response.status(status).json({
        type: this.getErrorType(status),
        title: this.getErrorTitle(status),
        status,
        detail: typeof exceptionResponse === 'string' 
          ? exceptionResponse 
          : (exceptionResponse as any).message,
        instance: request.url,
        requestId,
        timestamp: new Date().toISOString()
      });
      return;
    }

    // Handle Prisma errors
    if (exception instanceof PrismaClientKnownRequestError) {
      const prismaError = this.handlePrismaError(exception);
      response.status(prismaError.status).json({
        type: prismaError.type,
        title: prismaError.title,
        status: prismaError.status,
        detail: prismaError.detail,
        instance: request.url,
        requestId,
        timestamp: new Date().toISOString()
      });
      return;
    }

    // Handle unknown errors (don't leak internals)
    console.error('Unhandled exception:', exception);
    response.status(500).json({
      type: 'https://api.docita.work/errors/internal',
      title: 'Internal Server Error',
      status: 500,
      detail: 'An unexpected error occurred. Please try again later.',
      instance: request.url,
      requestId,
      timestamp: new Date().toISOString()
    });
  }

  private handlePrismaError(error: PrismaClientKnownRequestError) {
    switch (error.code) {
      case 'P2002': // Unique constraint violation
        return {
          type: 'https://api.docita.work/errors/conflict',
          title: 'Resource Conflict',
          status: 409,
          detail: `A resource with the same ${error.meta?.target} already exists`
        };
      case 'P2025': // Record not found
        return {
          type: 'https://api.docita.work/errors/not-found',
          title: 'Resource Not Found',
          status: 404,
          detail: 'The requested resource does not exist'
        };
      default:
        return {
          type: 'https://api.docita.work/errors/database',
          title: 'Database Error',
          status: 500,
          detail: 'A database error occurred'
        };
    }
  }

  private getErrorType(status: number): string {
    const types: Record<number, string> = {
      400: 'https://api.docita.work/errors/validation',
      401: 'https://api.docita.work/errors/unauthorized',
      403: 'https://api.docita.work/errors/forbidden',
      404: 'https://api.docita.work/errors/not-found',
      409: 'https://api.docita.work/errors/conflict',
      429: 'https://api.docita.work/errors/rate-limited',
      500: 'https://api.docita.work/errors/internal'
    };
    return types[status] || 'https://api.docita.work/errors/unknown';
  }

  private getErrorTitle(status: number): string {
    const titles: Record<number, string> = {
      400: 'Bad Request',
      401: 'Unauthorized',
      403: 'Forbidden',
      404: 'Not Found',
      409: 'Conflict',
      429: 'Too Many Requests',
      500: 'Internal Server Error'
    };
    return titles[status] || 'Unknown Error';
  }
}
```

## Validation Errors (class-validator)

```typescript
// DTO with validation
export class CreateAppointmentDto {
  @IsUUID()
  @IsNotEmpty()
  patientId: string;

  @IsUUID()
  @IsNotEmpty()
  clinicianId: string;

  @IsDateString()
  @IsNotEmpty()
  scheduledAt: string;

  @IsOptional()
  @IsString()
  @MaxLength(500)
  notes?: string;
}

// Validation pipe with custom error formatting
const validationPipe = new ValidationPipe({
  whitelist: true,
  forbidNonWhitelisted: true,
  transform: true,
  exceptionFactory: (errors) => {
    const formattedErrors: Record<string, string[]> = {};
    
    for (const error of errors) {
      const field = error.property;
      formattedErrors[field] = Object.values(error.constraints || {});
    }

    return new HttpException(
      {
        type: 'https://api.docita.work/errors/validation',
        title: 'Validation Error',
        status: 400,
        detail: 'The request body contains invalid fields',
        errors: formattedErrors
      },
      HttpStatus.BAD_REQUEST
    );
  }
});
```

## Healthcare-Specific Error Handling

```typescript
// PHI-safe error messages
function sanitizeError(error: any, user: User): string {
  // Never expose patient data in error messages
  if (error.message.includes('patient')) {
    return 'An error occurred while processing the request';
  }

  // Never expose database details
  if (error.message.includes('relation') || error.message.includes('column')) {
    return 'A database error occurred';
  }

  // Never expose internal IDs that could be used for enumeration
  if (error.message.includes('uuid') || error.message.includes('id')) {
    return 'The requested resource was not found';
  }

  return error.message;
}

// Audit failed access attempts
async function auditFailedAccess(
  req: Request,
  reason: string
): Promise<void> {
  await prisma.auditLog.create({
    data: {
      userId: req.user?.id,
      clinicId: req.user?.clinicId,
      action: 'ACCESS_DENIED',
      resource: req.url,
      method: req.method,
      reason,
      ipAddress: req.ip,
      userAgent: req.headers['user-agent'],
      timestamp: new Date()
    }
  });
}
```

## Error Response Patterns

### Batch Operations
```typescript
// Partial success in batch operations
interface BatchResponse<T> {
  successful: T[];
  failed: { item: any; error: ProblemDetails }[];
  summary: {
    total: number;
    succeeded: number;
    failed: number;
  };
}

// Example
const batchResult: BatchResponse<Appointment> = {
  successful: [appointment1, appointment2],
  failed: [
```

### Conditional Errors
```typescript
// ETag-based conflict detection
@Put(':id')
async update(
  @Param('id') id: string,
  @Body() data: UpdateAppointmentDto,
  @Headers('if-match') ifMatch: string
): Promise<Appointment> {
  const appointment = await this.service.findById(id);
  
  if (!appointment) {
    throw new NotFoundException('Appointment not found');
  }

  // Check ETag
  const currentTag = this.generateETag(appointment);
  if (ifMatch && ifMatch !== currentTag) {
    throw new ConflictException({
      type: 'https://api.docita.work/errors/stale-data',
      title: 'Stale Data',
      status: 409,
      detail: 'The resource has been modified since you last fetched it',
      currentETag: currentTag,
      receivedETag: ifMatch
    });
  }

  return this.service.update(id, data);
}
```

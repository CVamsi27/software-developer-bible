# Pagination — Cursor vs Offset, Keyset, Infinite Scroll

## Pagination Strategies

| Strategy | How It Works | Pros | Cons |
|----------|--------------|------|------|
| **Offset-based** | `?page=3&limit=20` → `OFFSET 40 LIMIT 20` | Simple, familiar | Slow on large offsets, unstable with inserts/deletes |
| **Cursor-based** | `?cursor=abc123&limit=20` → `WHERE id > 'abc123'` | Stable, performant | More complex, can't jump to page |
| **Keyset** | `?after=2024-03-15T10:00:00Z&limit=20` | Best for time-series | Requires indexed sort column |

## Offset-Based Pagination

```typescript
// Simple offset pagination
@Get()
async findAll(
  @Query('page') page: number = 1,
  @Query('limit') limit: number = 20
): Promise<PaginatedResponse<Appointment>> {
  const skip = (page - 1) * limit;
  
  const [data, total] = await Promise.all([
    prisma.appointment.findMany({
      where: { clinicId: req.user.clinicId },
      skip,
      take: limit,
      orderBy: { scheduledAt: 'desc' }
    }),
    prisma.appointment.count({
      where: { clinicId: req.user.clinicId }
    })
  ]);

  return {
    data,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit),
      hasNext: page * limit < total,
      hasPrev: page > 1
    }
  };
}
```

### Offset Pagination Problems
```typescript
// Problem 1: Inconsistent results with concurrent inserts
// Page 1: [A, B, C, D, E] (limit 5)
// Insert X between B and C
// Page 2: [D, E, F, G, H] (skips X!)

// Problem 2: Slow OFFSET on large tables
// OFFSET 1000000 LIMIT 20 → scans 1M rows, discards them

// Problem 3: Can't reliably resume after interruption
```

## Cursor-Based Pagination

```typescript
// Cursor pagination with Prisma
@Get()
async findAll(
  @Query('cursor') cursor?: string,
  @Query('limit') limit: number = 20
): Promise<CursorPaginatedResponse<Appointment>> {
  const take = limit + 1; // Fetch one extra to check for next page
  
  const data = await prisma.appointment.findMany({
    where: { clinicId: req.user.clinicId },
    take,
    ...(cursor && {
      cursor: { id: cursor },
      skip: 1 // Skip the cursor itself
    }),
    orderBy: { id: 'asc' }
  });

  const hasNext = data.length > limit;
  const items = hasNext ? data.slice(0, -1) : data;
  const nextCursor = hasNext ? items[items.length - 1].id : null;

  return {
    data: items,
    pagination: {
      nextCursor,
      hasNext,
      limit
    }
  };
}
```

### Cursor Encoding
```typescript
// Simple cursor (just the ID)
const simpleCursor = appointment.id;

// Opaque cursor (base64 encoded)
const opaqueCursor = Buffer.from(
  JSON.stringify({ id: appointment.id, timestamp: appointment.createdAt })
).toString('base64');

// Decode cursor
function decodeCursor(cursor: string): { id: string; timestamp: Date } {
  return JSON.parse(Buffer.from(cursor, 'base64').toString());
}

// Multi-field cursor (for compound sorting)
const multiFieldCursor = Buffer.from(
  JSON.stringify({
    scheduledAt: appointment.scheduledAt.toISOString(),
    id: appointment.id
  })
).toString('base64');
```

### Cursor Pagination Query
```typescript
// Prisma query with cursor
const appointments = await prisma.appointment.findMany({
  where: {
    clinicId: req.user.clinicId,
    // Cursor condition
    ...(cursor && {
      OR: [
        { scheduledAt: { lt: new Date(cursor.scheduledAt) } },
        {
          scheduledAt: new Date(cursor.scheduledAt),
          id: { gt: cursor.id }
        }
      ]
    })
  },
  take: limit + 1,
  orderBy: [
    { scheduledAt: 'desc' },
    { id: 'desc' }
  ]
});
```

## Keyset Pagination (Time-Series)

```typescript
// Keyset pagination for time-series data
@Get()
async findAll(
  @Query('after') after?: string, // ISO timestamp
  @Query('before') before?: string, // ISO timestamp
  @Query('limit') limit: number = 50
): Promise<KeysetPaginatedResponse<Appointment>> {
  const where: Prisma.AppointmentWhereInput = {
    clinicId: req.user.clinicId,
    scheduledAt: {}
  };

  if (after) {
    where.scheduledAt.gt = new Date(after);
  }
  if (before) {
    where.scheduledAt.lt = new Date(before);
  }

  const data = await prisma.appointment.findMany({
    where,
    take: limit + 1,
    orderBy: { scheduledAt: 'desc' }
  });

  const hasNext = data.length > limit;
  const items = hasNext ? data.slice(0, -1) : data;

  return {
    data: items,
    pagination: {
      nextAfter: items.length > 0 
        ? items[items.length - 1].scheduledAt.toISOString() 
        : null,
      hasNext,
      limit
    }
  };
}
```

## Infinite Scroll (Frontend Integration)

```typescript
// React Query infinite query
const useInfiniteAppointments = (clinicId: string) => {
  return useInfiniteQuery({
    queryKey: ['appointments', clinicId],
    queryFn: async ({ pageParam }) => {
      const params = new URLSearchParams({ limit: '20' });
      if (pageParam) {
        params.set('cursor', pageParam);
      }
      
      const response = await fetch(`/api/appointments?${params}`);
      return response.json();
    },
    getNextPageParam: (lastPage) => lastPage.pagination.nextCursor,
    initialPageParam: undefined
  });
};

// Virtual scrolling for large lists
import { useVirtualizer } from '@tanstack/react-virtual';

function AppointmentList() {
  const parentRef = useRef<HTMLDivElement>(null);
  const { data, fetchNextPage, hasNextPage } = useInfiniteAppointments(clinicId);

  const flatData = data?.pages.flatMap(page => page.data) ?? [];

  const virtualizer = useVirtualizer({
    count: hasNextPage ? flatData.length + 1 : flatData.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 80,
    overscan: 5
  });

  // Fetch more when near end
  useEffect(() => {
    const lastItem = virtualizer.getVirtualItems().at(-1);
    if (lastItem && lastItem.index >= flatData.length - 1 && hasNextPage) {
      fetchNextPage();
    }
  }, [virtualizer.getVirtualItems(), flatData.length, hasNextPage, fetchNextPage]);

  return (
    <div ref={parentRef} style={{ height: '600px', overflow: 'auto' }}>
      <div style={{ height: virtualizer.getTotalSize() }}>
        {virtualizer.getVirtualItems().map(virtualRow => (
          <AppointmentCard
            key={virtualRow.key}
            appointment={flatData[virtualRow.index]}
            style={{
              position: 'absolute',
              top: virtualRow.start,
              height: virtualRow.size
            }}
          />
        ))}
      </div>
    </div>
  );
}
```

## Pagination Comparison

| Criteria | Offset | Cursor | Keyset |
|----------|--------|--------|--------|
| **Performance** | Degrades with offset | Constant time | Constant time |
| **Stability** | Unstable with writes | Stable | Stable |
| **Jump to page** | ✅ Yes | ❌ No | ❌ No |
| **Total count** | ✅ Easy | ❌ Expensive | ❌ Expensive |
| **Implementation** | Simple | Medium | Medium |
| **Use case** | Small datasets | Large datasets | Time-series |

## When to Use Each

```typescript
const paginationStrategy = {
  // Use offset for:
  offset: [
    'Admin dashboards with page numbers',
    'Small datasets (<10K rows)',
    'When "jump to page" is required',
    'Export functionality (page through all)'
  ],

  // Use cursor for:
  cursor: [
    'Infinite scroll feeds',
    'Large datasets (>100K rows)',
    'Real-time data (appointments, notifications)',
    'Mobile apps (smooth scrolling)'
  ],

  // Use keyset for:
  keyset: [
    'Time-series data (logs, audit trails)',
    'Activity feeds',
    'Chat messages',
    'When natural sort key exists (created_at)'
  ]
};
```

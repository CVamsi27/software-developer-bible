# Advanced Redis Patterns — Distributed Locks, Rate Limiting, Pub/Sub, Streams, Lua Scripting

## 1. Distributed Locks (Redlock)

### The Problem
Multiple API instances might try to book the same appointment slot simultaneously. Without coordination, you get double-bookings.

### The Solution: Redlock Algorithm

```typescript
import Redlock from 'redlock';

const redlock = new Redlock(
  [redisClient],
  { retryCount: 3, retryDelay: 200 }
);

async function bookAppointment(
  slotId: string,
  dto: BookAppointmentDto
): Promise<Appointment> {
  // Acquire lock for this specific slot
  const lock = await redlock.acquire(
    [`lock:slot:${slotId}`],
    5000 // Lock expires in 5 seconds (safety net)
  );
  
  try {
    // Check slot availability (under lock)
    const slot = await prisma.slot.findUnique({ where: { id: slotId } });
    if (slot.booked) {
      throw new ConflictException('Slot already booked');
    }
    
    // Book the slot
    const appointment = await prisma.appointment.create({ data: dto });
    await prisma.slot.update({
      where: { id: slotId },
      data: { booked: true },
    });
    
    return appointment;
  } finally {
    // Always release the lock
    await lock.release();
  }
}
```

### When to Use Redlock vs SELECT FOR UPDATE

| Approach               | Use When                             | Trade-off                      |
| ----------               | ----------                             | -----------                      |
| **SELECT FOR UPDATE**  | Single database, pessimistic locking | Blocks other transactions      |
| **Redlock**            | Multiple instances, Redis available  | More complex, lock expiry risk |
| **Optimistic locking** | Low contention, version column       | Retry on conflict              |

## 2. Rate Limiting with Lua Scripts

### Token Bucket Algorithm

```lua
-- token_bucket.lua
local key = KEYS[1]
local max_tokens = tonumber(ARGV[1])
local refill_rate = tonumber(ARGV[2])
local now = tonumber(ARGV[3])

-- Get current state
local bucket = redis.call('HMGET', key, 'tokens', 'last_refill')
local tokens = tonumber(bucket[1]) or max_tokens
local last_refill = tonumber(bucket[2]) or now

-- Refill tokens based on elapsed time
local elapsed = now - last_refill
local new_tokens = math.min(max_tokens, tokens + elapsed * refill_rate)

-- Consume one token
if new_tokens >= 1 then
  new_tokens = new_tokens - 1
  redis.call('HMSET', key, 'tokens', new_tokens, 'last_refill', now)
  redis.call('EXPIRE', key, math.ceil(max_tokens / refill_rate) * 2)
  return 1 -- Allowed
else
  redis.call('HMSET', key, 'tokens', new_tokens, 'last_refill', now)
  return 0 -- Denied
end
```

### Implementation

```typescript
const TOKEN_BUCKET_SCRIPT = `
  local key = KEYS[1]
  local max_tokens = tonumber(ARGV[1])
  local refill_rate = tonumber(ARGV[2])
  local now = tonumber(ARGV[3])
  
  local bucket = redis.call('HMGET', key, 'tokens', 'last_refill')
  local tokens = tonumber(bucket[1]) or max_tokens
  local last_refill = tonumber(bucket[2]) or now
  
  local elapsed = now - last_refill
  local new_tokens = math.min(max_tokens, tokens + elapsed * refill_rate)
  
  if new_tokens >= 1 then
    new_tokens = new_tokens - 1
    redis.call('HMSET', key, 'tokens', new_tokens, 'last_refill', now)
    redis.call('EXPIRE', key, math.ceil(max_tokens / refill_rate) * 2)
    return 1
  else
    redis.call('HMSET', key, 'tokens', new_tokens, 'last_refill', now)
    return 0
  end
`;

async function checkRateLimit(
  clinicId: string,
  maxTokens: number = 100,
  refillRate: number = 1.67 // 100 tokens / 60 seconds
): Promise<{ allowed: boolean; remaining: number }> {
  const key = `ratelimit:${clinicId}`;
  const now = Date.now();
  
  const result = await redis.eval(
    TOKEN_BUCKET_SCRIPT,
    [key],
    [maxTokens, refillRate, now]
  );
  
  return {
    allowed: result === 1,
    remaining: await getRemainingTokens(key),
  };
}
```

## 3. Pub/Sub for Real-Time Events

### Basic Pub/Sub

```typescript
// Publisher
await redis.publish('clinic:123:appointment', JSON.stringify({
  type: 'APPOINTMENT_CREATED',
  appointmentId: 'abc-123',
  patientId: 'patient-456',
  timestamp: new Date().toISOString(),
}));

// Subscriber
const subscriber = redis.duplicate();
await subscriber.subscribe('clinic:123:appointment');

subscriber.on('message', (channel, message) => {
  const event = JSON.parse(message);
  console.log(`Received event: ${event.type}`);
  // Broadcast to WebSocket clients in this clinic room
  io.to(`clinic:123`).emit('appointment', event);
});
```

### Pattern-Based Subscription

```typescript
// Subscribe to all clinic events
await subscriber.psubscribe('clinic:*:appointment');

subscriber.on('pmessage', (pattern, channel, message) => {
  const clinicId = channel.split(':')[1];
  const event = JSON.parse(message);
  io.to(`clinic:${clinicId}`).emit('appointment', event);
});
```

## 4. Redis Streams for Event Sourcing

### Writing Events

```typescript
// Add event to stream
await redis.xadd(
  'appointments:stream',
  '*', // Auto-generate ID
  'type', 'APPOINTMENT_CREATED',
  'appointmentId', 'abc-123',
  'patientId', 'patient-456',
  'clinicId', 'clinic-789',
  'timestamp', new Date().toISOString()
);
```

### Reading Events (Consumer Groups)

```typescript
// Create consumer group
try {
  await redis.xgroup('CREATE', 'appointments:stream', 'workers', '0');
} catch (e) {
  // Group already exists
}

// Read events as consumer
const events = await redis.xreadgroup(
  'GROUP', 'workers', 'worker-1',
  'COUNT', 10,
  'BLOCK', 5000,
  'STREAMS', 'appointments:stream', '>'
);

for (const [stream, messages] of events) {
  for (const [id, fields] of messages) {
    const event = parseStreamFields(fields);
    await processEvent(event);
    
    // Acknowledge event
    await redis.xack('appointments:stream', 'workers', id);
  }
}
```

## 5. Caching Patterns

### Cache-Aside (Lazy Loading)

```typescript
async function getPatient(id: string): Promise<Patient> {
  // 1. Check cache
  const cached = await redis.get(`patient:${id}`);
  if (cached) {
    return JSON.parse(cached);
  }
  
  // 2. Cache miss — fetch from DB
  const patient = await prisma.patient.findUnique({ where: { id } });
  
  // 3. Populate cache (with TTL)
  await redis.setex(
    `patient:${id}`,
    300, // 5 minutes TTL
    JSON.stringify(patient)
  );
  
  return patient;
}
```

### Write-Through

```typescript
async function updatePatient(id: string, data: PatientUpdate): Promise<Patient> {
  // 1. Write to DB
  const patient = await prisma.patient.update({ where: { id }, data });
  
  // 2. Update cache immediately
  await redis.setex(
    `patient:${id}`,
    300,
    JSON.stringify(patient)
  );
  
  return patient;
}
```

### Cache Invalidation

```typescript
// Delete cache on write
async function deletePatient(id: string): Promise<void> {
  await prisma.patient.delete({ where: { id } });
  
  // Invalidate cache
  await redis.del(`patient:${id}`);
  
  // Also invalidate related caches
  await redis.del(`patient:${id}:appointments`);
  await redis.del(`clinic:${clinicId}:patients`);
}
```

## 6. Lua Scripting Best Practices

### Atomic Operations

```lua
-- Credit wallet atomically
local wallet_key = KEYS[1]
local amount = tonumber(ARGV[1])
local operation = ARGV[2] -- 'credit' or 'debit'

local balance = tonumber(redis.call('GET', wallet_key)) or 0

if operation == 'debit' then
  if balance < amount then
    return {err = 'Insufficient balance'}
  end
  balance = balance - amount
else
  balance = balance + amount
end

redis.call('SET', wallet_key, balance)
return balance
```

### Idempotent Operations

```lua
-- Process event exactly once
local event_key = KEYS[1]
local event_id = ARGV[1]

-- Check if already processed
if redis.call('SISMEMBER', 'processed_events', event_id) == 1 then
  return 0 -- Already processed
end

-- Mark as processed
redis.call('SADD', 'processed_events', event_id)
redis.call('EXPIRE', 'processed_events', 86400) -- 24h TTL

return 1 -- Newly processed
```

## Interview Tips

- **Q: When would you use Redis over PostgreSQL for caching?**
  - Redis: sub-millisecond reads, ephemeral data, rate limiting
  - PostgreSQL: persistent data, complex queries, ACID compliance

- **Q: How do you prevent cache stampede?**
  - Locking (setnx) during cache refresh
  - Probabilistic early expiration
  - Stale-while-revalidate pattern

- **Q: What's the difference between Redis Pub/Sub and Streams?**
  - Pub/Sub: fire-and-forget, no persistence
  - Streams: persistent, consumer groups, acknowledgment

# CI/CD Pipeline Patterns — GitHub Actions, GitLab CI, Deployment Strategies

## GitHub Actions (NestJS + Prisma)

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    
    services:
      postgres:
        image: postgres:15
        env:
          POSTGRES_DB: test_db
          POSTGRES_USER: test_user
          POSTGRES_PASSWORD: test_pass
        ports:
          - 5432:5432
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5

      redis:
        image: redis:7
        ports:
          - 6379:6379

    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Run Prisma migrations
        run: npx prisma migrate deploy
        env:
          DATABASE_URL: postgresql://test_user:test_pass@localhost:5432/test_db
      
      - name: Run linting
        run: npm run lint
      
      - name: Run type checking
        run: npm run typecheck
      
      - name: Run unit tests
        run: npm run test
      
      - name: Run integration tests
        run: npm run test:e2e
        env:
          DATABASE_URL: postgresql://test_user:test_pass@localhost:5432/test_db
          REDIS_URL: redis://localhost:6379
      
      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          token: ${{ secrets.CODECOV_TOKEN }}

  build:
    needs: test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Build
        run: npm run build
      
      - name: Build Docker image
        run: docker build -t api:${{ github.sha }} .
      
      - name: Push to registry
        run: |
          echo ${{ secrets.DOCKER_PASSWORD }} | docker login -u ${{ secrets.DOCKER_USERNAME }} --password-stdin
          docker tag api:${{ github.sha }} registry.example.com/api:${{ github.sha }}
          docker push registry.example.com/api:${{ github.sha }}
```

## Deployment Pipeline

```yaml
# .github/workflows/deploy.yml
name: Deploy

on:
  push:
    branches: [main]

jobs:
  deploy-staging:
    runs-on: ubuntu-latest
    environment: staging
    steps:
      - name: Deploy to staging
        run: |
          kubectl set image deployment/api api=registry.example.com/api:${{ github.sha }} -n staging
          kubectl rollout status deployment/api -n staging
      
      - name: Run smoke tests
        run: |
          curl -f https://staging.api.example.com/health
          npm run test:smoke -- --base-url=https://staging.api.example.com

  deploy-production:
    needs: deploy-staging
    runs-on: ubuntu-latest
    environment: production
    steps:
      - name: Deploy to production (canary)
        run: |
          # Deploy to 10% of pods
          kubectl set image deployment/api api=registry.example.com/api:${{ github.sha }} -n production
          kubectl rollout status deployment/api -n production
      
      - name: Monitor error rate
        run: |
          # Wait 5 minutes, check error rate
          sleep 300
          ERROR_RATE=$(curl -s "http://prometheus:9090/api/v1/query?query=rate(http_requests_total{status=~'5..'}[5m])" | jq '.data.result[0].value[1]')
          if (( $(echo "$ERROR_RATE > 0.01" | bc -l) )); then
            echo "Error rate too high, rolling back"
            kubectl rollout undo deployment/api -n production
            exit 1
          fi
      
      - name: Complete rollout
        run: |
          # If monitoring passed, continue rollout
          kubectl rollout status deployment/api -n production
```

## GitLab CI (NestJS)

```yaml
# .gitlab-ci.yml
stages:
  - test
  - build
  - deploy

variables:
  POSTGRES_DB: test_db
  POSTGRES_USER: test_user
  POSTGRES_PASSWORD: test_pass

test:
  stage: test
  image: node:20-alpine
  services:
    - postgres:15
    - redis:7
  script:
    - npm ci
    - npx prisma migrate deploy
    - npm run lint
    - npm run typecheck
    - npm run test
    - npm run test:e2e
  coverage: '/Lines\s*:\s*(\d+\.?\d*)%/'

build:
  stage: build
  image: docker:latest
  services:
    - docker:dind
  script:
    - docker build -t $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA .
    - docker push $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA
  only:
    - main

deploy-staging:
  stage: deploy
  script:
    - kubectl set image deployment/api api=$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA -n staging
  environment:
    name: staging
  only:
    - main

deploy-production:
  stage: deploy
  script:
    - kubectl set image deployment/api api=$CI_REGISTRY_IMAGE:$CI_COMMIT_SHA -n production
  environment:
    name: production
  when: manual
  only:
    - main
```

## Deployment Strategies

```typescript
const deploymentStrategies = {
  // Rolling update
  rolling: {
    description: 'Gradually replace old pods with new ones',
    pros: ['Simple', 'No extra infrastructure'],
    cons: ['Slow rollback', 'Mixed versions during deploy'],
    k8s: 'strategy.type: RollingUpdate'
  },

  // Blue-green
  blueGreen: {
    description: 'Run two identical environments, switch traffic',
    pros: ['Instant rollback', 'No downtime'],
    cons: ['Double infrastructure cost', 'State management'],
    k8s: 'Two deployments + service selector'
  },

  // Canary
  canary: {
    description: 'Route small percentage of traffic to new version',
    pros: ['Low risk', 'Real-world testing'],
    cons: ['Complex routing', 'Longer deploy time'],
    k8s: 'Istio VirtualService weight-based routing'
  },

  // Feature flags
  featureFlags: {
    description: 'Deploy code disabled, enable via flag',
    pros: ['Instant enable/disable', 'A/B testing'],
    cons: ['Code complexity', 'Flag management'],
    k8s: 'Same deployment, feature toggle in code'
  }
};
```

## Interview Tips
- Explain CI/CD pipeline stages (lint → test → build → deploy)
- Know deployment strategies (rolling, blue-green, canary)
- Discuss caching strategies for faster builds
- Explain security scanning in pipelines
- Know rollback strategies (automatic on error rate spike)
